\chapter{Mathematical Derivation Details}
\label{app:derivations}

\section{Sutton-Graves Heat Flux Derivation}

Sutton-Graves stagnation heat flux~\cite{sutton1971gravesh}:
\begin{equation}
    \dot{q} = C_{SG} \sqrt{\frac{\rho_\infty}{R_N}} \, V_\infty^3
    \label{eq:sg_derivation}
\end{equation}

\noindent where $C_{SG} = 1.83 \times 10^{-4}$\,W\,m$^{-2}$\,(kg\,m$^{-3}$)$^{-1/2}$\,(m\,s$^{-1}$)$^{-3}$ is the 5-species air coefficient, giving the $V^3$ dependence.

\subsection{Dimensional Verification}

With $\rho_\infty = 7.851 \times 10^{-4}$\,kg/m$^3$ (ISA at 51.8\,km), $R_N = 1.5$\,m, $V_\infty = 3378$\,m/s:
\begin{align}
    \dot{q} &= 1.83 \times 10^{-4} \times \sqrt{\frac{7.851 \times 10^{-4}}{1.5}} \times (3378)^3 \nonumber \\
    &= 1.83 \times 10^{-4} \times \sqrt{5.234 \times 10^{-4}} \times 3.8552 \times 10^{10} \nonumber \\
    &= 1.83 \times 10^{-4} \times 0.022879 \times 3.8552 \times 10^{10} \nonumber \\
    &\approx 1.614 \times 10^{5} \text{ W/m}^2 = 16.14 \text{\,W/cm}^2
    \label{eq:sg_baseline}
\end{align}

\noindent The value is $+12.4\%$ on the $14.36$\,W/cm$^2$ flight measurement~\cite{old2013irve3postflight} and $+5.8\%$ on Rapisarda's $15.26$\,W/cm$^2$ peak~\cite{rapisarda2023mdao}; the denser MCD v6.1 trajectory nearly offsets the single ISA point. The engine comments report $12.20$\,W/cm$^2$ from the defaults $C_{SG} = 1.7415 \times 10^{-4}$, $R_N = 0.55$\,m and $V_\infty = 2700$\,m/s, not the run condition.

\section{Mean Free Path and Knudsen Number}

Hard-sphere mean free path~\cite{bird1994molecular}:
\begin{equation}
    \lambda = \frac{1}{\sqrt{2} \, \pi \, d^2 \, n}
    \label{eq:mfp_derivation}
\end{equation}

\noindent where $d = 3.7 \times 10^{-10}$\,m is the molecular diameter and $n$ the number density:
\begin{equation}
    Kn = \frac{\lambda}{L}
    \label{eq:kn_derivation}
\end{equation}

\noindent where $L$ is the aeroshell diameter $D = 3.0$\,m.

\section{Fay-Riddell Stagnation-Point Heat Transfer}

The Fay-Riddell equation~\cite{fay1958stagnation} gives the stagnation heat flux of a reacting, dissociated boundary layer:
\begin{equation}
    \dot{q}_s = 0.763 \, \text{Pr}^{-0.6} \, (\rho_e \mu_e)^{0.4} \, (\rho_w \mu_w)^{0.1} \, \sqrt{\left(\frac{du_e}{dx}\right)_s} \, (h_s - h_w)
    \label{eq:fay_riddell}
\end{equation}

\noindent Unlike Sutton-Graves, Fay-Riddell adds shock layer structure, chemical non-equilibrium and catalytic wall effects.
