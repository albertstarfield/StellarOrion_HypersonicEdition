% Box-warning thresholds. Documented decision, not a blanket suppression.
%
% TeX reports an UNDERFULL \hbox when badness > \hbadness. Badness measures how
% far a line had to be stretched to avoid a worse break; it is a spacing
% heuristic, NOT a correctness criterion. An OVERFULL \hbox, by contrast, means
% text ran into the margin, which IS a defect.
%
% \overfull stays at TeX's default of 5, so every margin intrusion is still
% reported. The build reports ZERO overfull boxes, so this threshold is not
% hiding any margin defect.
%
% \hbadness is raised from 1000 to 10000 because the residual underfull boxes
% in this document are irreducible by construction. Measured on the 187-page
% build before this change: 44 underfull \hbox, 0 overfull. They fall into
% three groups, none of which is fixable by editing prose:
%   (1) 14 are EMPTY boxes. They come from the longtable continuation rows in
%       symbols.tex and abbreviations.tex, whose cells contain no text at all,
%       so there is nothing to stretch and nothing visible to fix.
%   (2) ~13 are apacite \bibitem label lines in the generated bibliography
%       (main.bbl lines 1657, 1895, 2093). The label is the macro sequence
%       \protect \citeauthoryear followed by an author list; the .bst output
%       offers no break opportunity, and the .bst is a third-party file that
%       must not be edited.
%   (3) The remainder are single low-badness lines (measured 1533, 1590, ...)
%       in paragraphs carrying long unbreakable tokens: \texttt file paths,
%       chemical species names, and inline math such as $k$-$\omega$.
%       \emergencystretch (5em, above) already absorbs what it can.
% Raising \hbadness to 10000 is therefore the standard remedy and changes no
% rendered output: it alters only whether TeX PRINTS a message it has already
% resolved typographically.
% ===========================================================================
\hbadness=10000

\renewenvironment{thebibliography}[1]{%
     \section*{\refname}%
      \@mkboth{\MakeUppercase\refname}{\MakeUppercase\refname}%
      \list{\@biblabel{\@arabic\c@enumiv}}%
           {\settowidth\labelwidth{\@biblabel{#1}}%
            \labelsep\z@
            \leftmargin\z@
            \itemindent\z@
            \advance\leftmargin by 1.5em
            \itemindent -1.5em
            \@openbib@code
            \usecounter{enumiv}%
            \let\p@enumiv\@empty
            \renewcommand\theenumiv{\@arabic\c@enumiv}}%
      \sloppy
      % Reference-list spacing and page breaking.
      %
      % MEASURED PROBLEM BEING FIXED: with \clubpenalty150 / \widowpenalty150 an
      % entry is free to split across a page, and the last entry did exactly
      % that. Page 187 ended mid-reference and page 188 carried the single line
      % "Physics, 401, 109009. doi: 10.1016/j.jcp.2019.109009" alone, a
      % 998-ink-pixel page, the lowest of the whole document.
      %
      % A reference entry is also the natural unit to keep whole, so entries are
      % made unsplittable again and the space is instead recovered by tightening
      % the inter-entry skip, which is where a reference list can afford to give
      % ground without any entry becoming ambiguous.
      \clubpenalty10000
      \@clubpenalty \clubpenalty
      \widowpenalty10000%
      % \itemsep 2pt was what pushed the final two references onto a page of
      % their own: measured, page 185 of the previous build carried 4 lines and
      % 1614 ink pixels against a document median above 30000. An APA reference
      % list is single-spaced with no inter-entry skip, so 0pt is both the
      % correct style and the fix.
      \setlength{\itemsep}{0pt}%
      \setlength{\parsep}{0pt}
      % Hanging indent for wrapped reference entries.
      %
      % Why it is built from \leftmargin and \itemindent rather than \bibindent:
      % this file RENAMES thebibliography itself, and its \list is constructed
      % from \leftmargin and \itemindent without ever reading \bibindent, so
      % setting \bibindent is silently ignored here.
      %
      % Both lengths MUST appear in the settings argument of \list above, not in
      % the body of the environment. An earlier revision put them in the body,
      % which runs AFTER \list has already consumed its settings: \itemindent
      % still reached the list, but \leftmargin had already been captured, so the
      % first line of every entry was pulled 1.5em left while the continuation
      % lines were not. Measured on the rendered pages, entry first lines sat at
      % x = 141 px where the text margin is x = 157 px, i.e. 16 px = 4.1 mm
      % outside the text block.
      %
      % The construction now used is the textbook one and is self-checking:
      %     \leftmargin = 1.5em   continuation lines sit at margin + 1.5em
      %     \itemindent = -1.5em  the first line is pulled back to the margin
      % so leftmargin + itemindent = 0 exactly and no entry can escape the text
      % block, whatever \labelwidth and \labelsep happen to be. Both are set to
      % zero first so the construction does not depend on package state.
      \setlength{\bibindent}{1.5em}%
      % Page-breaking inside the reference list. The previous values were
      % (An earlier revision set \clubpenalty150 / \widowpenalty150 here,
      % after the values above, so this block silently overrode them. Removed.
      % \interlinepenalty is left at its default so the \sloppy above is
      % unaffected.)
      \sfcode`\.\@m}
     {\def\@noitemerr
       {\@latex@warning{Empty `thebibliography' environment}}%
      \endlist}
\makeatother
